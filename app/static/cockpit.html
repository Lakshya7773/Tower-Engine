1. Google Sheets Custom Function: =TOWER_FEASIBILITY(lat, lon)

function TOWER_FEASIBILITY(lat, lon) {
  var url = "__ORIGIN__/api/v2/feasibility/analyze";
  var res = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify({ lat: lat, lon: lon, search_radius_m: 500 })
  });
  var data = JSON.parse(res.getContentText());
  return [[
    data.verdict,
    data.overall_score,
    data.engines.typology_and_structural.data.asset_typology,
    data.engines.financial_boq.data.capex_boq_lakh.total_capex_lakh,
    data.engines.financial_boq.data.irr_sensitivity["1.5x"].irr_10yr_pct
  ]];
}
